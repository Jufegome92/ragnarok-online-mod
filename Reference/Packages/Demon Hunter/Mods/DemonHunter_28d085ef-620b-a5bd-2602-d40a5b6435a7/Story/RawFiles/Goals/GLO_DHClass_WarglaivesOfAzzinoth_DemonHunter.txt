// DH_DemonHunter_Class also forces the DemonHunter Tag as a boost to help the trigger.

THEN

// Since LevelUP doesn't cover Leveling Down.

// ---- Riddle for Bell Tower Solve, adding here just for reference incase I want to do more with this.
// At the tallest peak where whispers conspire,
// Three tolls of the bell will summon the fire.
// Where chaos swells deep, and green flames ignite,
// Awakening a weapon that sows fear and destruction in fight.

//-- Force user to find the bell and ring it 3x to be awarded the loot.
// Increment the bell counter.
PROC
PROC_DHClassWarglaivesAzzinothRingBell_Increment((ITEM)_Item)
AND
DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Item,_CurrentCounterValue)
AND
IntegerSum(_CurrentCounterValue, 1, _IncrementedCounterValue)
THEN
NOT DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Item,_CurrentCounterValue);  // Remove Old DB
DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Item,_IncrementedCounterValue);  // Set a New DB Value

// Initialize the DB counter to track number of rings on the bell.
IF
TemplateUseStarted(_Character,TOOL_GEN_Tower_Bell_A_6051a6d2-88e9-4ab4-999c-f463632453eb,_Bell)
AND
NOT DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Bell,_)
THEN
DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Bell,0);

// Increment the bell if the quest was completed and we have not been awarded.
IF
TemplateUseFinished(_Character,TOOL_GEN_Tower_Bell_A_6051a6d2-88e9-4ab4-999c-f463632453eb,_Bell,1)
AND
NOT QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_Circus_RingMaster_Complete((GUIDSTRING)_Character)
THEN
PROC_DHClassWarglaivesAzzinothRingBell_Increment((ITEM)_Bell);

// Increment Bell If Lucretious was Killed and we have not been awarded.
IF
TemplateUseFinished(_Character,TOOL_GEN_Tower_Bell_A_6051a6d2-88e9-4ab4-999c-f463632453eb,_Bell,1)
AND
NOT QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_CircusRingMaster_IsPermaDead((GUIDSTRING)_Character)
THEN
PROC_DHClassWarglaivesAzzinothRingBell_Increment((ITEM)_Bell);

// Award at the 3rd ring and reset the counter.
PROC
PROC_DHClassWarglaivesAzzinoth_MH_Award((GUIDSTRING)_Character,(ITEM)_Bell)
AND
DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Bell,_CurrentCounterValue)
AND
_CurrentCounterValue >= 3
THEN
NOT DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Bell,_CurrentCounterValue); // Remove Old DB
DB_DHClassWarglaivesAzzinothRingMasterBellCounter_DemonHunter(_Bell,0);                        // Reset Counter
PROC_DHClassWarglaivesAzzinoth_PlayAcquisitionEffect(_Bell);
TemplateAddTo((ITEMROOT)WPN_DHGW_AzzinothWarglaives_MH_d4f1073e-710e-4cdc-bd90-b9fe73a2389e, (GUIDSTRING)_Character, 1, 1);
PROC_DH_WarglaiveAzzinothMainHand_SetFlagAwarded((GUIDSTRING)_Character); // Flag we were rewarded

IF
TemplateUseFinished(_Character,TOOL_GEN_Tower_Bell_A_6051a6d2-88e9-4ab4-999c-f463632453eb,_Bell,1)
AND
NOT QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_Circus_RingMaster_Complete((GUIDSTRING)_Character)
THEN
PROC_DHClassWarglaivesAzzinoth_MH_Award((GUIDSTRING)_Character,(ITEM)_Bell);

// Increment Bell If Lucretious was Killed and we have not been awarded.
IF
TemplateUseFinished(_Character,TOOL_GEN_Tower_Bell_A_6051a6d2-88e9-4ab4-999c-f463632453eb,_Bell,1)
AND
NOT QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_CircusRingMaster_IsPermaDead((GUIDSTRING)_Character)
THEN
PROC_DHClassWarglaivesAzzinoth_MH_Award((GUIDSTRING)_Character,(ITEM)_Bell);

//--- - MAIN HAND AWARD BUG FIX START (10/15/2024)
// Version 1.1.0.10 had a bug with locking out the main hand award if the offhand was awarded first
// This was because I was flagging the MH as a reward instead of the OH.  It was also
// Causing the offhand to continously be rewarded each time the puzzle is retriggered.
QRY
QRY_DH_WarglaiveAzzinoth_WasAwardedMH_FLAGGED_BUG((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_WasAwardedOffHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
// Check if we flagged current session as bug fixed
GetFlag((FLAG)DH_LOOT_Rewarded_WarglaivesOfAzzinoth_MH_RewardBugFix_03f552c7-0585-407a-8201-00f7bf48582d, (GUIDSTRING)_Character, _wasFixed)
AND
_wasFixed == 1
THEN
DB_NOOP(1);

// Bug Fix Prep
// We need to ensure that we flag the OH as awarded if it exists in the user's inventory
// Since this wasn't getting flagged, we should flag this first on SavegameLoad before we even
// do anything else.
IF
SavegameLoaded()
AND
GetHostCharacter(_Character)
AND
TemplateIsInInventory((ITEMROOT)WPN_DHGW_AzzinothWarglaives_OH_a4f629f5-12df-4746-bf55-7a494e6c64c1, (GUIDSTRING)_Character, 1)
THEN
PROC_DH_WarglaiveAzzinothOffHand_SetFlagAwarded(_Character);

// Bug Fix Logic
// Check if we have been flagged as fixed (Bail out if true)
// Since the bug entailed the offhand acquisition was flagging the MH as being acquired
// We can check the if OH and MH was awarded and the MH weapon is NOT in the inventory
// This means we have a bugged save.  Remove the MH flag aquisition, so the player
// Can re-ring the bell to acquire the weapon.
IF
SavegameLoaded()
AND
GetHostCharacter(_Character)
AND
NOT QRY_DH_WarglaiveAzzinoth_WasAwardedMH_FLAGGED_BUG((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_WasAwardedOffHand((GUIDSTRING)_Character)
AND
QRY_DH_WarglaiveAzzinoth_WasAwardedMainHand((GUIDSTRING)_Character)
AND
TemplateIsInInventory((ITEMROOT)WPN_DHGW_AzzinothWarglaives_MH_d4f1073e-710e-4cdc-bd90-b9fe73a2389e, (GUIDSTRING)_Character, _Count)
AND
_Count == 0  // Not found in inventory
THEN
// Remove the MH Awarded Flag
ClearFlag((FLAG)DH_LOOT_Rewarded_WarglaivesOfAzzinoth_MH_83f84de2-ebbe-4408-9d3b-d63a533b2f21, _Character);
// Flag that we have fixed the issue to avoid further extra acquistion.
// NOTE: There is a possibility that the user will end up with 2 MH if they load the game
// and the MH is not in their inventory, e.g. drooped on floor while loading the save
//  but thats ok at least we can retro fix previous saves.
SetFlag((FLAG)DH_LOOT_Rewarded_WarglaivesOfAzzinoth_MH_RewardBugFix_03f552c7-0585-407a-8201-00f7bf48582d, _Character);


//--- - MAIN HAND AWARD BUG FIX END
