//--  Handle Weapon Race Variation Swaps BEGIN
// We are doing weapon look variaion for some races
// Becuase some of them hold the weapons on idle, so aggressively
// So the models are adjusted to counteract the animation.
PROC  // Githyanki
PROC_DH_WeaponModelVariation_DoGithSwap((GUIDSTRING)_Character, (ITEM)_Item)
AND
QRY_DH_IsDemonHunter(_Character)
AND
GetRace((CHARACTER)_Character, 1, _Race)
AND
_Race == "Githyanki"
AND
GetTemplate(_Item, _Template)
AND
DB_DHClassWeaponModelVariation_DemonHunter(_Template,_ModelVersion)
THEN
Transform(_Item, _ModelVersion, (SHAPESHIFTRULE)PHYSICAL_4acc6277-6dcd-4110-9450-b9379beaedac);

IF 
Equipped(_Item, _Character)
THEN
PROC_DH_WeaponModelVariation_DoGithSwap(_Character, _Item);

IF
SavegameLoaded()
THEN
IterateCharacters("DH_Class_WeaponSwapCheck", "DH_Class_WeaponSwapCheck_Completed");

IF
LongRestFinished()
THEN
IterateCharacters("DH_Class_WeaponSwapCheck", "DH_Class_WeaponSwapCheck_Completed");

IF 
RestorePartyFinished()
THEN
IterateCharacters("DH_Class_WeaponSwapCheck", "DH_Class_WeaponSwapCheck_Completed");

IF
EntityEvent(_Character,"DH_Class_WeaponSwapCheck")
THEN
PROC_DH_TriggerWeaponEvent((GUIDSTRING)_Character);
